// HÀM TIỆN ÍCH XỬ LÝ NGÀY THÁNG (Cho tính năng SCOPE)
// ==========================================
function parseDateStr(dateStr) {
  if (!dateStr || dateStr.trim() === "") return null;
  
  // Hỗ trợ định dạng YYYY-MM-DD nếu lỡ có lưu sai trước đó
  if (dateStr.indexOf('-') !== -1) {
    var parts = dateStr.split('-');
    if (parts.length === 3) return new Date(parts[0], parts[1] - 1, parts[2]);
  }
  
  // Định dạng chuẩn DD/MM/YYYY
  var parts = dateStr.split('/');
  if (parts.length === 3) return new Date(parts[2], parts[1] - 1, parts[0]);
  
  return null;
}

function formatDateStr(dateObj) {
  if (!dateObj) return "";
  var dd = String(dateObj.getDate()).padStart(2, '0');
  var mm = String(dateObj.getMonth() + 1).padStart(2, '0');
  var yyyy = dateObj.getFullYear();
  return dd + '/' + mm + '/' + yyyy;
}

// API LẤY & LƯU TRỮ DEADLINE
// ==========================================
function getDeadlinesUser(mssv) {
  var sheetName = getDeadlineSheetName(mssv);
  var sheet = ss.getSheetByName(sheetName);
  var result = [];
  
  // Lấy Deadline cá nhân hiện tại
  if (sheet) {
    var data = sheet.getDataRange().getDisplayValues();
    if (data.length > 1) {
      for (var i = 1; i < data.length; i++) {
        if (String(data[i][0]).trim() === String(mssv).trim()) {
          var row = [];
          for (var col = 0; col < 8; col++) row.push(col < data[i].length ? data[i][col] : "");
          row.push(i + 1); 
          result.push(row);
        }
      }
    }
  }

  // TRỘN THÊM CÁC HỌC PHẦN HỆ THỐNG CÓ GHI 'VLE'
  var regSheet = ss.getSheetByName("UserRegisteredCourses");
  var masterSheet = ss.getSheetByName("MasterTKB");

  if (regSheet && masterSheet) {
    var regData = regSheet.getDataRange().getDisplayValues();
    var registeredIds = [];
    
    for (var j = 1; j < regData.length; j++) {
      if (String(regData[j][0]).trim() === String(mssv).trim()) {
        registeredIds = String(regData[j][1]).split(',').map(function(id) { return id.trim(); }).filter(Boolean);
        break;
      }
    }

    if (registeredIds.length > 0) {
      var masterData = masterSheet.getDataRange().getDisplayValues();
      for (var k = 1; k < masterData.length; k++) {
        var courseId = String(masterData[k][0]).trim(); 
        if (registeredIds.indexOf(courseId) !== -1) {
          
          var hinhThuc = String(masterData[k][5]).toUpperCase();
          var phong = String(masterData[k][7]).toUpperCase();
          var tenMon = String(masterData[k][6]).toUpperCase();
          
          // Chỉ lấy nếu thuộc VLE
          if (hinhThuc.indexOf("VLE") !== -1 || phong.indexOf("VLE") !== -1 || tenMon.indexOf("VLE") !== -1) {
            var mon = masterData[k][6] || "";
            var ngayBD = masterData[k][10] || "";
            var ngayKT = masterData[k][11] || "";
            var thoiGianDesc = (ngayBD && ngayKT) ? "Từ " + ngayBD + " đến " + ngayKT : "Học phần trực tuyến";
            
            result.push([
              mssv,                  // [0]
              mon,                   // [1] Tiêu đề
              thoiGianDesc,          // [2] Thời lượng
              "VLE",        // [3] Nhãn Tag
              "primary",             // [4] Icon nền (Màu xanh dương)
              "🌐",                   // [5] Emoji
              ngayBD,                // [6] Ngày BĐ
              ngayKT,                // [7] Ngày KT
              "SYS_VLE_" + courseId  // [8] Gắn mã ID hệ thống
            ]);
          }
        }
      }
    }
  }
  
  return result;
}
function addDeadlineUser(mssv, title, duration, tag, icon, emoji, startDate, endDate) {
  var sheetName = getDeadlineSheetName(mssv);
  var sheet = ss.getSheetByName(sheetName);
  if(!sheet) {
    sheet = ss.insertSheet(sheetName);
    sheet.appendRow(["MSSV", "Title", "Duration", "Tag", "Icon", "Emoji", "StartDate", "EndDate"]);
  }
  sheet.appendRow([mssv, title, duration, tag, icon, emoji, startDate, endDate]);
  return "Đã thêm Deadline thành công!";
}

function editDeadlineUser(rowIndex, mssv, title, duration, tag, icon, emoji, startDate, endDate) {
  var sheetName = getDeadlineSheetName(mssv);
  var sheet = ss.getSheetByName(sheetName);

  sheet.getRange(rowIndex, 1, 1, 8).setValues([[mssv, title, duration, tag, icon, emoji, startDate, endDate]]);
  return "Cập nhật Deadline thành công!";
}

function deleteDeadlineUser(rowIndex, mssv) {
  var sheetName = getDeadlineSheetName(mssv);
  var sheet = ss.getSheetByName(sheetName);
  if(sheet) sheet.deleteRow(rowIndex);
  return "Đã xóa Deadline!";
}


function registerUser(mssv, password, name) {
  var sheet = ss.getSheetByName("Users");
  var data = sheet.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][0]).trim() === String(mssv).trim()) {
      return { success: false, message: "Tài khoản mã số sinh viên này đã đăng ký trước đó!" };
    }
  }
  sheet.appendRow([mssv, password, name]);
  return { success: true, message: "Đăng ký thành công!" };
}

function changePassword(mssv, oldPassword, newPassword) {
  var sheet = ss.getSheetByName("Users");
  var data = sheet.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][0]).trim() === String(mssv).trim()) {
      if (String(data[i][1]).trim() === String(oldPassword).trim()) {
        sheet.getRange(i + 1, 2).setValue(newPassword);
        return "Thay đổi mật khẩu cá nhân thành công!";
      } else {
        return "Lỗi: Mật khẩu cũ hiện tại nhập không chính xác.";
      }
    }
  }
  return "Lỗi: Không tìm thấy thông tin tài khoản người dùng.";
}

          if (hinhThuc.indexOf("VLE") === -1 && phong.indexOf("VLE") === -1 && mon.indexOf("VLE") === -1) {

 var gsSheet = ss.getSheetByName("GroupStudyBooking");

          if (regList[r].split(':::')[0].trim() === String(mssv).trim()) {
              isJoined = true;
              break;
          }

      if (isJoined) {

          gsData[g][7],  // ThoiGian

function pingPresence(uuid, mssv, lastView) {
  var lock = LockService.getScriptLock();
  var onlineList = [];
  
  if (lock.tryLock(2000)) {
    try {
      var props = PropertiesService.getScriptProperties();
      var now = Date.now();
      
      var userInfo = { time: now, mssv: mssv || "Khách" };
      props.setProperty('user_' + uuid, JSON.stringify(userInfo));
      
      if (mssv && mssv !== "Khách" && mssv.indexOf("|") !== -1) {
        var cleanMssv = mssv.split("|")[0].trim();
        updateLastActive(cleanMssv, lastView);
      }
      
      var allProps = props.getProperties();
      var keysToDelete = [];
      
      for (var key in allProps) {
        if (key.indexOf('user_') === 0) {
          try {
            var data = JSON.parse(allProps[key]);

            if (now - data.time > 90000) { 
              keysToDelete.push(key); 
            } else { 
              if (onlineList.indexOf(data.mssv) === -1) {
                  onlineList.push(data.mssv);
              }
            }
          } catch(e) { 
            keysToDelete.push(key); 
          }
        }
      }
      keysToDelete.forEach(function(k) { props.deleteProperty(k); });
    } finally {
      lock.releaseLock();
    }
  }
  return { count: onlineList.length, list: onlineList };
}

function updateLastActive(mssv, lastView) {
  try {
    var sheet = ss.getSheetByName("Users");
    if (!sheet) return;
    
    var data = sheet.getDataRange().getValues();
    var now = new Date();
    var nowStr = Utilities.formatDate(now, "GMT+7", "yyyy-MM-dd HH:mm:ss");
    
    for (var i = 1; i < data.length; i++) {
      if (String(data[i][0]).trim() === String(mssv).trim()) {
        var oldLastActiveStr = data[i][8];   // Cột I (Index 8) - Lần cuối
        var oldCurrentView = data[i][9];     // Cột J (Index 9) - Đang xem
        var oldPrevView1 = data[i][10];      // Cột K (Index 10) - Trước 1
        var oldPrevView2 = data[i][11];      // Cột L (Index 11) - Trước 2
        var oldFirstAccessStr = data[i][13]; // Cột N (Index 13) - Lần đầu đợt
        
        // BỘ LỌC KHỬ RÁC (Chỉ xóa nếu là Object hoặc đúng định dạng YYYY-MM-DD)
        var dateRegex = /^\d{4}-\d{2}-\d{2}/; 
        if (typeof oldCurrentView === 'object' || dateRegex.test(String(oldCurrentView))) oldCurrentView = "";
        if (typeof oldPrevView1 === 'object' || dateRegex.test(String(oldPrevView1))) oldPrevView1 = "";
        if (typeof oldPrevView2 === 'object' || dateRegex.test(String(oldPrevView2))) oldPrevView2 = "";

        // TÍNH TOÁN LOGIC 30 PHÚT ĐỢT MỚI ĐÃ ĐƯỢC SỬA LỖI TIMEZONE
        var isNewSession = true;
        if (oldLastActiveStr) {
          var timeNowFixed = new Date(nowStr.replace(' ', 'T') + "Z").getTime();
          var oldStrFixed = String(oldLastActiveStr).trim();
          if (typeof oldLastActiveStr === 'object' && oldLastActiveStr instanceof Date) {
             oldStrFixed = Utilities.formatDate(oldLastActiveStr, "GMT+7", "yyyy-MM-dd HH:mm:ss");
          }
          var timeOldFixed = new Date(oldStrFixed.replace(' ', 'T') + "Z").getTime();
          if (!isNaN(timeOldFixed) && !isNaN(timeNowFixed)) {
            var diff = timeNowFixed - timeOldFixed;
            if (diff >= 0 && diff < 30 * 60 * 1000) isNewSession = false;
          }
        }
        
        // 1. Cập nhật "Lần cuối truy cập" (Cột I)
        sheet.getRange(i + 1, 9).setValue(nowStr);
        
       
        // 2. Đẩy lịch sử: Đang xem -> Trước 1 -> Trước 2 -> Trước 3
        if (lastView && String(lastView) !== String(oldCurrentView)) {
          var newStr = String(lastView).trim();
          var oldStr = String(oldCurrentView).trim();
          
          sheet.getRange(i + 1, 10).setValue(newStr); // J: Đang xem luôn được cập nhật
          
          if (oldCurrentView) {
            if (newStr !== oldStr) {
                var parentView = newStr.indexOf(" - ") !== -1 ? newStr.split(" - ")[0].trim() : newStr;
                var oldParentView = oldStr.indexOf(" - ") !== -1 ? oldStr.split(" - ")[0].trim() : oldStr;
                
                // TRƯỜNG HỢP 1: Dọn rác trạng thái chờ (LOẠI BỎ CHỮ THỜI GIAN BIỂU BỊ DƯ)
                // Nếu bước cũ CHỈ LÀ danh mục gốc của bước mới (VD: "Thời gian biểu" -> "Thời gian biểu - Tuần 15")
                if (oldStr === parentView && newStr !== oldStr) {
                    // => Không làm gì cả. Chữ "Thời gian biểu" trơ trọi sẽ bị ĐÈ MẤT bởi newStr ở cột J phía trên, 
                    // mà không bị đẩy xuống cột K (Phía trước 1).
                }
                // TRƯỜNG HỢP 2: Nội suy bước trung gian
                // Chuyển từ một danh mục KHÁC hoàn toàn thẳng vào chi tiết (VD: "Trang chủ" -> "Thời gian biểu - Tuần 8")
                else if (parentView !== newStr && oldStr !== parentView && parentView !== oldParentView) {
                    sheet.getRange(i + 1, 11).setValue(parentView); // K: Chèn danh mục gốc vào Trước 1
                    sheet.getRange(i + 1, 12).setValue(oldStr);     // L: Đẩy trang cũ xuống Trước 2
                    if (oldPrevView1 && oldPrevView1 !== oldStr) {
                        sheet.getRange(i + 1, 13).setValue(oldPrevView1); // M: Trước 3
                    }
                } 
                // TRƯỜNG HỢP 3: Đẩy tuần tự bình thường 
                // Ghi nhận chuyển qua lại giữa các tuần (VD: "Thời gian biểu - Tuần 6" -> "Thời gian biểu - Tuần 8")
                else {
                    sheet.getRange(i + 1, 11).setValue(oldStr); // K: Trước 1
                    if (oldPrevView1 && oldPrevView1 !== oldStr) {
                        sheet.getRange(i + 1, 12).setValue(oldPrevView1); // L: Trước 2
                        if (oldPrevView2) {
                            sheet.getRange(i + 1, 13).setValue(oldPrevView2); // M: Trước 3
                        }
                    }
                }
            }
          }
        } else if (lastView && !oldCurrentView) {
          sheet.getRange(i + 1, 10).setValue(lastView);
        }
        
        // 3. Cập nhật "Lần đầu truy cập cho đợt" (Cột N)
        // Lần này logic đợt mới isNewSession đã hoạt động đúng
        if (isNewSession || !oldFirstAccessStr || String(oldFirstAccessStr).trim() === "") {
          sheet.getRange(i + 1, 14).setValue(nowStr);
        }
        
        break;
      }
    }
  } catch (e) {
    Logger.log("Lỗi updateLastActive: " + e.toString());
  }
}

    var shouldBeNew = false;
    var hasCondition = false;
    
    // 1. Quét tìm Deadline ở các cột 2, 3, 7
    var deadlineTime = extractDeadlineGAS(c2) || extractDeadlineGAS(c3) || extractDeadlineGAS(c7);
    
    if (deadlineTime) {
      hasCondition = true;
      if (now <= deadlineTime) {
        shouldBeNew = true;
      }
    } else {
      // 2. Không có Deadline -> Dựa vào Ngày đăng bài
      var postDateMatch = c4.match(/(\d{1,2})\/(\d{1,2})\/(\d{4})/);
      if (postDateMatch) {
        hasCondition = true;
        var postDate = new Date(postDateMatch[3], postDateMatch[2] - 1, postDateMatch[1]).getTime();
        var daysPassed = (now - postDate) / (1000 * 60 * 60 * 24);
        
        if (daysPassed <= 15) {
          shouldBeNew = true; 
        }
      }
    }
    
    var finalC1 = c1;
    
    // XỬ LÝ LOGIC CHỮ "NEW"
    if (sheetName.toLowerCase() === 'thông báo') {
      if (hasCondition) {
        if (shouldBeNew && !isCurrentlyNew) {
          finalC1 = c1 === "" ? "New" : "New - " + c1;
          hasChanges = true;
        } 
        else if (!shouldBeNew && isCurrentlyNew) {
          finalC1 = c1.replace(/\bnew\b/ig, '').trim(); 
          finalC1 = finalC1.replace(/^[\s\-]+|[\s\-]+$/g, ''); 
          hasChanges = true;
        }
      }
    } else {
      if (isCurrentlyNew) {
         finalC1 = c1.replace(/\bnew\b/ig, '').trim(); 
         finalC1 = finalC1.replace(/^[\s\-]+|[\s\-]+$/g, '');
         hasChanges = true;
      }
    }
    
    // Cập nhật vào mảng kết quả trả về cho Web
    data[i][0] = finalC1;
    // Cập nhật vào mảng chờ để ghi xuống Sheet
    col1Updates.push([finalC1]);
  }
  
  // NẾU PHÁT HIỆN SỰ THAY ĐỔI -> GHI XUỐNG SHEET 1 LẦN DUY NHẤT (SIÊU TỐC)
  if (hasChanges) {
    sheet.getRange(2, 1, col1Updates.length, 1).setValues(col1Updates);
  }
  
  return data;
}

function extractDeadlineGAS(text) {
  if (!text) return null;
  
  // Biểu thức quét siêu linh hoạt:
  // - Bắt được chữ "Ngày", "Ng&agrave;y", "Ngày." hoặc "Ng&agrave;y."
  // - Hoặc không cần chữ Ngày nào luôn cũng bắt được
  var match = text.match(/DEADLINE[\s=:]*(?:lúc\s*)?(?:(\d{1,2}:\d{2}))?[\s]*(?:Ng(?:à|&agrave;)y\.?\s*)?(\d{1,2}\/\d{1,2}\/\d{4})/i);
  
  if (match) {
    var timeStr = match[1] || "23:59"; // Mặc định 23:59 nếu không nhập giờ
    var dateStr = match[2];            // Lấy chuỗi ngày tháng
    
    var dateParts = dateStr.split('/');
    var timeParts = timeStr.split(':');
    
    // Quy đổi ra mili-giây để tính toán
    return new Date(dateParts[2], dateParts[1] - 1, dateParts[0], timeParts[0], timeParts[1], 0).getTime();
  }
  return null;
}

function moveSheetRow(sheetName, rowIndex, direction) {
  var sheet = ss.getSheetByName(sheetName);
  var lastRow = sheet.getLastRow();
  if (direction === 'up' && rowIndex <= 2) return "Đã ở vị trí trên cùng.";
  if (direction === 'down' && rowIndex >= lastRow) return "Đã ở vị trí dưới cùng.";
  var targetIndex = (direction === 'up') ? rowIndex - 1 : rowIndex + 2;
  sheet.insertRowBefore(targetIndex);
  var sourceRange = sheet.getRange((direction === 'up') ? rowIndex + 1 : rowIndex, 1, 1, sheet.getMaxColumns());
  var targetRange = sheet.getRange(targetIndex, 1);
  sourceRange.moveTo(targetRange);
  sheet.deleteRow((direction === 'up') ? rowIndex + 1 : rowIndex);
  return "Đã di chuyển hàng thành công.";
}

function getStudentEmailByMssv(mssv) {
  try {
    var sheet = ss.getSheetByName("Users");
    if (!sheet) return null;
    var data = sheet.getDataRange().getValues();
    if (data.length <= 1) return null;
    
    // Tìm vị trí cột "Email liên hệ" từ dòng tiêu đề (dòng đầu tiên)
    var headers = data[0];
    var emailColIndex = -1;
    for (var h = 0; h < headers.length; h++) {
      var headerText = String(headers[h]).trim().toLowerCase();
      if (headerText === "email liên hệ" || headerText === "email") {
        emailColIndex = h;
        break;
      }
    }
    
    // Nếu không tìm thấy cột email, mặc định lấy cột F (index 5) theo hình của bạn
    if (emailColIndex === -1) emailColIndex = 5; 

    // Duyệt tìm email của sinh viên theo MSSV (cột A tương ứng index 0)
    for (var i = 1; i < data.length; i++) {
      if (String(data[i][0]).trim() === String(mssv).trim()) {
        var email = String(data[i][emailColIndex] || "").trim();
        if (email.indexOf("@") !== -1) {
          return email;
        }
      }
    }
  } catch (e) {
    Logger.log("Lỗi tra cứu email sinh viên: " + e.toString());
  }
  return null;
}

function getSystemTKBList(mssv) {

  // 1. Lấy danh sách toàn bộ môn học từ hệ thống

  var allCourses = [];

  for (var i = 1; i < data.length; i++) {
    if (data[i][0]) {
      allCourses.push({
        id: data[i][0],
        thu: data[i][1],
        tietBd: parseInt(data[i][2]),
        soTiet: parseInt(data[i][3]),
        thoiGian: data[i][4],
        hinhThuc: data[i][5],
        mon: data[i][6],
        phong: data[i][7],
        gv: data[i][8],
        ngayBatDau: data[i][10] || "", // Cột K
        ngayKetThuc: data[i][11] || "", // Cột L

        hocKy: data[i][13] || "",        // Cột N (Dùng cho dropdown Lọc Học Kỳ)
        ngayNgoaiLe: data[i][14] || "",
        ghiChu: data[i][15] || ""
      });
    }
  }

  // 2. Lấy danh sách ID mà sinh viên đã đăng ký
  var regSheet = ss.getSheetByName("UserRegisteredCourses");
  var registeredIds = [];
  if (regSheet) {
    var regData = regSheet.getDataRange().getDisplayValues();

}

function undoVote(rowIndex, voteType, user) {
    var qaSheet = ss.getSheetByName("Q&A");
    var colIndex = (voteType === 'up') ? 5 : 6; 
    
    var cell = qaSheet.getRange(rowIndex, colIndex);
    var currentVotes = parseInt(cell.getValue()) || 0;
    
    // Trừ đi 1 điểm, không để số điểm nhỏ hơn 0
    cell.setValue(Math.max(0, currentVotes - 1));

    var totalUp = parseInt(qaSheet.getRange(rowIndex, 5).getValue()) || 0;
    var totalDown = parseInt(qaSheet.getRange(rowIndex, 6).getValue()) || 0;
    
    return { up: totalUp, down: totalDown };
}

function assignStudentsToCourse(courseId, assignedMssvList) {
  var regSheet = ss.getSheetByName("UserRegisteredCourses");
  var data = regSheet.getDataRange().getValues();
  var mssvArray = assignedMssvList.split(',').map(function(item) { return item.trim(); }).filter(Boolean);
  
  // Xóa courseId khỏi tất cả sinh viên trước, sau đó chỉ thêm lại cho những người được check
  for (var i = 1; i < data.length; i++) {
    var mssv = String(data[i][0]).trim();
    var courses = String(data[i][1]).split(',').map(function(c) { return c.trim(); }).filter(Boolean);
    
    var index = courses.indexOf(String(courseId).trim());
    var isAssigned = mssvArray.indexOf(mssv) !== -1;
    
    if (isAssigned && index === -1) {
      courses.push(courseId); // Thêm môn
      regSheet.getRange(i + 1, 2).setValue(courses.join(','));
    } else if (!isAssigned && index !== -1) {
      courses.splice(index, 1); // Rút môn
      regSheet.getRange(i + 1, 2).setValue(courses.join(','));
    }
  }
  return "Đã cập nhật danh sách sinh viên tham gia học phần!";
}

function assignStudentsToCourse(courseId, assignedMssvList) {
  var regSheet = ss.getSheetByName("UserRegisteredCourses");
  if (!regSheet) {
    regSheet = ss.insertSheet("UserRegisteredCourses");
    regSheet.appendRow(["MSSV", "CourseIDs"]);
  }
  var data = regSheet.getDataRange().getValues();
  var mssvArray = assignedMssvList.split(',').map(function(item) { return item.trim(); }).filter(Boolean);
  
  var existingMssvs = [];
  for (var i = 1; i < data.length; i++) {
    var mssv = String(data[i][0]).trim();
    existingMssvs.push(mssv);
    var courses = String(data[i][1]).split(',').map(function(c) { return c.trim(); }).filter(Boolean);
    
    var index = courses.indexOf(String(courseId).trim());
    var isAssigned = mssvArray.indexOf(mssv) !== -1;
    
    if (isAssigned && index === -1) {
      courses.push(courseId); // Gán thêm môn
      regSheet.getRange(i + 1, 2).setValue(courses.join(','));
    } else if (!isAssigned && index !== -1) {
      courses.splice(index, 1); // Rút bớt môn
      regSheet.getRange(i + 1, 2).setValue(courses.join(','));
    }
  }
  
  // Xử lý thêm sinh viên hoàn toàn mới (chưa từng có hàng dữ liệu nào trước đây)
  for (var j = 0; j < mssvArray.length; j++) {
    if (existingMssvs.indexOf(mssvArray[j]) === -1) {
      regSheet.appendRow([mssvArray[j], String(courseId).trim()]);
    }
  }
  return "Đã cập nhật danh sách sinh viên tham gia học phần!";
}
function dragDropSheetRow(sheetName, fromIndex, toIndex) {
  var sheet = ss.getSheetByName(sheetName);
  if (fromIndex === toIndex) return "Không có sự thay đổi vị trí.";
  
  // Xác định dòng đích để chèn khoảng trống
  var targetIndex = toIndex > fromIndex ? toIndex + 1 : toIndex;
  sheet.insertRowBefore(targetIndex);
  
  // Do vừa chèn thêm 1 dòng, vị trí dòng gốc có thể bị đẩy xuống
  var actualFromIndex = toIndex < fromIndex ? fromIndex + 1 : fromIndex;
  var sourceRange = sheet.getRange(actualFromIndex, 1, 1, sheet.getMaxColumns());
  var targetRange = sheet.getRange(targetIndex, 1);
  
  // Di chuyển dữ liệu và xóa dòng cũ
  sourceRange.moveTo(targetRange);
  sheet.deleteRow(actualFromIndex);
  
  return "Đã kéo thả và sắp xếp thành công!";
}

function syncToGoogleCalendar(events) {
  try {
    var cal = CalendarApp.getDefaultCalendar();
    var addedCount = 0;
    
    for (var i = 0; i < events.length; i++) {
      var ev = events[i];
      var calEvent = null;
      
      var success = false;
      var retries = 0;
      var maxRetries = 4; // Tối đa thử lại 4 lần cho 1 sự kiện
      
      // Vòng lặp thử lại nếu bị Google chặn do Rate Limit
      while (!success && retries < maxRetries) {
        try {
          // Xử lý sự kiện cả ngày (Deadline)
          if (ev.type === 'ALLDAY') {
            var startDate = new Date(ev.startDate);
            var endDate = new Date(ev.endDate);
            startDate.setHours(0, 0, 0, 0);
            endDate.setHours(0, 0, 0, 0);
            
            if (startDate.getTime() === endDate.getTime()) {
              calEvent = cal.createAllDayEvent(ev.title, startDate, {
                location: ev.location || "",
                description: ev.description || ""
              });
            } else {
              var adjustedEndDate = new Date(endDate);
              adjustedEndDate.setDate(adjustedEndDate.getDate() + 1);
              calEvent = cal.createAllDayEvent(ev.title, startDate, adjustedEndDate, {
                location: ev.location || "",
                description: ev.description || ""
              });
            }
          } 
          // Xử lý sự kiện theo giờ (Có Vòng lặp hoặc Sự kiện đơn)
          else {
            if (ev.type === 'RECURRING') {
              var recurrence = CalendarApp.newRecurrence().addWeeklyRule().times(ev.count);
              calEvent = cal.createEventSeries(ev.title, new Date(ev.start), new Date(ev.end), recurrence, {
                location: ev.location || "",
                description: ev.description || ""
              });
            } else {
              calEvent = cal.createEvent(ev.title, new Date(ev.start), new Date(ev.end), {
                location: ev.location || "",
                description: ev.description || ""
              });
            }
          }
          
// ĐẶT CHUÔNG BÁO THEO ĐÚNG QUY LUẬT TÍNH TOÁN TỪ FRONTEND
// Đã mở khóa cho cả sự kiện lặp (RECURRING)
if (calEvent && ev.reminders && ev.reminders.length > 0) {
    try {
        calEvent.removeAllReminders(); // Xóa các thông báo mặc định của tài khoản
        for (var r = 0; r < ev.reminders.length; r++) {
            calEvent.addPopupReminder(ev.reminders[r]); // Chèn giờ thông báo chính xác
        }
    } catch(e) {
        // Ignored: Bỏ qua nếu Google quá tải hoặc lỗi thêm nhắc nhở
    }
}
          success = true; // Đánh dấu thành công để thoát vòng lặp while
          addedCount++;
          
          // Tạm nghỉ 1.5 giây giữa mỗi sự kiện để không bị Google đánh dấu spam
          Utilities.sleep(1500); 

        } catch(e) {
          var errorMsg = e.toString().toLowerCase();
          // Nếu gặp lỗi do tạo quá nhanh
          if (errorMsg.indexOf("quá nhiều") !== -1 || errorMsg.indexOf("too many") !== -1 || errorMsg.indexOf("limit") !== -1) {
            retries++;
            if (retries >= maxRetries) {
              throw e; // Ném lỗi ra ngoài nếu thử quá số lần nhưng vẫn bị chặn
            }
            // Ngủ lâu hơn 1 chút trước khi thử lại (3s, 6s, 9s...)
            Utilities.sleep(retries * 3000);
          } else {
            throw e; // Các lỗi khác không phải do spam thì ném ra ngay
          }
        }
      }
    }
    
    return "Tuyệt vời! Đã đồng bộ trực tiếp " + addedCount + " sự kiện vào Google Calendar của bạn.";
  } catch(err) {
    return "Lỗi đồng bộ: " + err.toString() + ". (Lưu ý: Bạn hãy đợi khoảng 15-30 phút để Google mở khóa chống Spam rồi hẵng thử lại nhé).";
  }
}

function toggleGroupStudyReg(p) {
  var sheet = ss.getSheetByName("GroupStudyBooking");
  if (!sheet) return "Lỗi: Không tìm thấy cơ sở dữ liệu.";
  
  var data = sheet.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    if (data[i][0] === p.id) {
      var maxSlot = parseInt(data[i][13]);
      var registeredStr = data[i][15] ? String(data[i][15]) : "";
      var regList = registeredStr.split(',').filter(function(x) { return x; });
      
      // Tìm vị trí MSSV trong mảng đã nối chuỗi
      var index = -1;
      for (var r = 0; r < regList.length; r++) {
          if (regList[r].split(':::')[0].trim() === p.mssv) {
              index = r;
              break;
          }
      }
      
      if (p.type === 'register') {
        if (regList.length >= maxSlot) return "Lỗi: Lịch học nhóm đã đầy slot!";
        if (index === -1) {
            // Nối MSSV và Ghi chú bằng chuỗi phân cách ':::'
            var userNote = p.note ? ":::" + p.note : ":::";
            regList.push(p.mssv + userNote);
        }
      } else if (p.type === 'cancel') {
        if (index !== -1) regList.splice(index, 1);
      }
      
      sheet.getRange(i + 1, 16).setValue(regList.join(','));
      return p.type === 'register' ? "Đã đăng ký và đồng bộ vào TKB!" : "Đã hủy đăng ký khỏi TKB!";
    }
  }
  return "Lỗi: Không tìm thấy lịch nhóm này.";
}

function loginUser(mssv, password) {
  var sheet = ss.getSheetByName("Users");
  if (!sheet) return { success: false, message: "Lỗi: Không tìm thấy bảng dữ liệu Users." };
  
  var data = sheet.getDataRange().getValues();
  
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][0]).trim() === String(mssv).trim()) {
      if (String(data[i][1]).trim() === String(password).trim()) {
        
        // Kiểm tra xem sinh viên đã làm khảo sát chưa
        var isSurveyDone = false;
        var ksSheet = ss.getSheetByName("KhaoSat");
        if (ksSheet) {
          var ksData = ksSheet.getDataRange().getValues();
          for (var j = 1; j < ksData.length; j++) {
            if (String(ksData[j][1]).trim() === String(mssv).trim()) {
              isSurveyDone = true;
              break;
            }
          }
        }
        
        return {
          success: true,
          mssv: data[i][0],
          name: data[i][2],
          chuyenNganh: data[i][3] || "",
          khoa: data[i][4] || "",
          khoaHoc: data[i][5] || "",
          nhom: data[i][6] || "",
          avatar: data[i][7] || "",
          isSurveyDone: isSurveyDone
        };
      } else {
        return { success: false, message: "Mật khẩu không chính xác!" };
      }
    }
  }
  return { success: false, message: "Tài khoản không tồn tại trên hệ thống!" };
}
